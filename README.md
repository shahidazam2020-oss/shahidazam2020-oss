<script>

const form = document.getElementById("feedbackForm");
const stars = document.querySelectorAll(".star");
const ratingText = document.getElementById("ratingText");
const alertBox = document.getElementById("alert");
const submitBtn = document.getElementById("submitBtn");
const card = document.getElementById("card");
const thanks = document.getElementById("thanks");
const given = document.getElementById("given");

const labels = [
  "Poor",
  "Fair",
  "Good",
  "Very Good",
  "Excellent"
];

function updateStars(value){

  stars.forEach((star,index)=>{
    star.classList.toggle("lit",index < value);
  });

  if(value > 0){
    ratingText.textContent =
      labels[value-1] + " — " + value + " out of 5";
  }else{
    ratingText.textContent = "Select a star to rate";
  }
}

stars.forEach((star,index)=>{

  const input = star.querySelector("input");

  star.addEventListener("mouseenter",()=>{
    updateStars(index + 1);
  });

  input.addEventListener("change",()=>{
    updateStars(index + 1);
  });

});

document.getElementById("stars").addEventListener("mouseleave",()=>{

  const selected =
    document.querySelector('input[name="Rating"]:checked');

  updateStars(selected ? Number(selected.value) : 0);

});

form.addEventListener("submit",function(e){

  const selected =
    document.querySelector('input[name="Rating"]:checked');

  alertBox.classList.remove("show");

  if(!selected){

    e.preventDefault();

    alertBox.textContent =
      "Please select a star rating before submitting.";

    alertBox.classList.add("show");

    return;
  }

  const email =
    document.getElementById("email").value.trim();

  if(email &&
     !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)){

    e.preventDefault();

    alertBox.textContent =
      "Please enter a valid email address.";

    alertBox.classList.add("show");

    return;
  }

  /*
    FormSubmit requires the first submission to activate
    the email address. After activation, submissions work normally.
  */

  submitBtn.disabled = true;
  submitBtn.textContent = "Submitting...";

  /*
    GitHub Pages-safe success page.
    The form is submitted normally to FormSubmit.
  */

  const currentURL =
    window.location.href.split("?")[0];

  document.getElementById("nextPage").value =
    currentURL + "?submitted=1";

});

if(new URLSearchParams(window.location.search).get("submitted") === "1"){

  const savedRating =
    sessionStorage.getItem("rating");

  if(savedRating){

    card.classList.add("done");

    given.textContent =
      "★".repeat(Number(savedRating)) +
      "☆".repeat(5 - Number(savedRating));

  }

}

form.addEventListener("submit",()=>{

  const selected =
    document.querySelector('input[name="Rating"]:checked');

  if(selected){
    sessionStorage.setItem(
      "rating",
      selected.value
    );
  }

});

document.getElementById("again").addEventListener("click",()=>{

  sessionStorage.removeItem("rating");

  form.reset();

  updateStars(0);

  card.classList.remove("done");

  submitBtn.disabled = false;

  submitBtn.textContent = "Submit Feedback";

  window.history.replaceState(
    {},
    document.title,
    window.location.pathname
  );

});

</script>
