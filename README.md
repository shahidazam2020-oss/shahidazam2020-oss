      s.style.setProperty("--x", Math.cos(a) * d + "px");
      s.style.setProperty("--y", Math.sin(a) * d * .8 + 60 + "px");
      s.style.setProperty("--r", (Math.random() * 720 - 360) + "deg");
      s.style.background = colors[i % colors.length];
      s.style.animationDelay = Math.random() * .12 + "s";
      if (i % 3 === 0) s.style.borderRadius = "50%";
      box.appendChild(s);
    }
    setTimeout(() => (box.innerHTML = ""), 2000);
  }

  function setLoading(on) {
    btn.disabled = on; btn.classList.toggle("loading", on);
    btnText.textContent = on ? "Sending..." : "Submit Feedback";
  }

  /* ---------- submit ---------- */
  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    alertBox.classList.remove("show");

    if (form._honey.value) return;                       // bot caught by the honeypot

    const rating = current();
    if (!rating) {
      starsWrap.classList.remove("shake"); void starsWrap.offsetWidth; starsWrap.classList.add("shake");
      showError("Please choose a star rating before submitting.");
      return;
    }

    const email = form.email.value.trim();
    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      showError("That email address does not look right. Fix it or leave it empty.");
      form.email.focus(); return;
    }

    let last = 0; try { last = Number(localStorage.getItem("fb:last") || 0); } catch (_) {}
    if (Date.now() - last < CONFIG.cooldownMs) {
      showError("Thanks, you just sent a response. Please wait a minute before sending another.");
      return;
    }

    const data = {
      rating,
      liked: form.liked.value.trim(),
      suggestions: form.suggestions.value.trim(),
      name: form.name.value.trim(),
      email
    };

    setLoading(true);
    try {
      const res = await fetch(ENDPOINT, {
        method: "POST",
        headers: { "Content-Type": "application/json", "Accept": "application/json" },
        body: JSON.stringify({
          _subject: `${CONFIG.subjectPrefix}: ${"★".repeat(rating)}${"☆".repeat(5 - rating)} (${rating}/5)`,
          _template: "table",
          _captcha: "false",
          _replyto: email || undefined,
          "Rating": `${rating} / 5 (${LABELS[rating - 1]})`,
          "What they liked": data.liked || "Not provided",
          "Suggestions": data.suggestions || "Not provided",
          "Name": data.name || "Anonymous",
          "Email": email || "Not provided",
          "Sent from": location.href
        })
      });
      const out = await res.json().catch(() => ({}));
      if (!res.ok || String(out.success) === "false") throw new Error(out.message || "Request failed");

      try { localStorage.setItem("fb:last", String(Date.now())); } catch (_) {}
      $("#given").textContent = "★".repeat(rating) + "☆".repeat(5 - rating);
      card.classList.add("done");
      window.scrollTo({ top: 0, behavior: "smooth" });
      confetti();
    } catch (err) {
      showError(`Could not send right now. Please try again, or <a href="${mailtoFallback(data)}">send it by email instead</a>.`);
    } finally {
      setLoading(false);
    }
  });

  $("#again").addEventListener("click", () => {
    form.reset(); paint(0); describe(0);
    document.querySelectorAll(".count").forEach(c => c.textContent = "0 / 600");
    card.classList.remove("done");
  });
})();
</script>
